-- =========================================================================
-- 3. GATING SETTINGS
-- Config fields are locked once referenced by a video session. is_active can
-- still be flipped to select the current rule; create a new row for new rules.
-- =========================================================================
CREATE TABLE gating_settings (
    id                          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    model_version_id            UUID NOT NULL REFERENCES model_versions(id) ON DELETE RESTRICT,
    confidence_threshold        NUMERIC(5,4) NOT NULL,
    temporal_window_ms          INTEGER NOT NULL,
    minimum_detection_count     INTEGER,
    aggregation_method          gating_aggregation_method NOT NULL DEFAULT 'CONSECUTIVE',
    frame_sampling_fps          NUMERIC(6,2),
    is_active                   BOOLEAN NOT NULL DEFAULT false,
    notes                       TEXT,
    created_at                  TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT chk_gating_confidence CHECK (confidence_threshold BETWEEN 0 AND 1),
    CONSTRAINT chk_gating_window CHECK (temporal_window_ms > 0),
    CONSTRAINT chk_gating_min_count CHECK (
        minimum_detection_count IS NULL OR minimum_detection_count >= 1
    ),
    CONSTRAINT chk_gating_required_count CHECK (
        aggregation_method = 'ROLLING_AVERAGE'
        OR minimum_detection_count IS NOT NULL
    ),
    CONSTRAINT chk_gating_fps CHECK (frame_sampling_fps IS NULL OR frame_sampling_fps > 0)
);

CREATE UNIQUE INDEX uq_gating_one_active_per_model
    ON gating_settings (model_version_id) WHERE is_active = true;

-- =========================================================================
-- 8. AWARENESS RESPONSES
-- =========================================================================
-- Only authenticated users may save awareness answers. The application creates one
-- assessment_session_id per participant's PRE/POST cycle and reuses it for both stages.
CREATE TABLE awareness_responses (
    id                      UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    assessment_session_id   UUID NOT NULL,
    user_id                 UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    question_id             UUID NOT NULL REFERENCES awareness_questions(id) ON DELETE RESTRICT,
    response_value          JSONB NOT NULL,
    is_correct              BOOLEAN,
    submitted_at            TIMESTAMPTZ NOT NULL DEFAULT now(),
    UNIQUE (assessment_session_id, question_id)
);

CREATE INDEX idx_awareness_responses_user_session
    ON awareness_responses (user_id, assessment_session_id, submitted_at DESC);

-- Prevent a client from reusing one assessment-session UUID across different users.
CREATE OR REPLACE FUNCTION fn_validate_awareness_session_owner()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
    IF EXISTS (
        SELECT 1
        FROM awareness_responses ar
        WHERE ar.assessment_session_id = NEW.assessment_session_id
          AND ar.user_id <> NEW.user_id
          AND ar.id <> NEW.id
    ) THEN
        RAISE EXCEPTION 'An awareness assessment session can belong to only one user';
    END IF;
    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_awareness_session_owner
    BEFORE INSERT OR UPDATE ON awareness_responses
    FOR EACH ROW EXECUTE FUNCTION fn_validate_awareness_session_owner();

    gating_setting_id           UUID REFERENCES gating_settings(id) ON DELETE RESTRICT,

    tracker_version             VARCHAR(50),

    gate_decision               decision_status,
    failure_reason              TEXT,

    decision_confidence         NUMERIC(5,4),

        estimated_distinct_tracks IS NULL OR estimated_distinct_tracks >= 0

        decision_confidence IS NULL OR decision_confidence BETWEEN 0 AND 1
    ),
    CONSTRAINT chk_video_processing_duration CHECK (
        processing_duration_ms IS NULL OR processing_duration_ms >= 0

    CONSTRAINT chk_video_fps CHECK (achieved_fps IS NULL OR achieved_fps >= 0),
    CONSTRAINT chk_video_end_time CHECK (ended_at IS NULL OR ended_at >= started_at),

        media_sha256 IS NULL OR media_sha256 ~ '^[0-9a-fA-F]{64}$'

        gate_decision IS NULL OR processing_status = 'COMPLETED'
    ),
    CONSTRAINT chk_video_completed_requires_gate CHECK (
        processing_status <> 'COMPLETED'
        OR (gate_decision IS NOT NULL AND gating_setting_id IS NOT NULL)

    CONSTRAINT chk_video_failure_reason CHECK (
        failure_reason IS NULL OR processing_status = 'FAILED'
    )

    CONSTRAINT chk_track_avg_confidence CHECK (
        avg_confidence IS NULL OR avg_confidence BETWEEN 0 AND 1

-- =========================================================================
-- 11. DETECTIONS
-- YOLO-native normalized center coordinates (cx, cy, width, height).
-- =========================================================================
CREATE TABLE detections (
    id                  UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    video_session_id    UUID NOT NULL REFERENCES video_sessions(id) ON DELETE CASCADE,
    track_id            UUID,
    frame_number        INTEGER NOT NULL,
    timestamp_ms        INTEGER NOT NULL,
    bbox_cx             NUMERIC(7,6) NOT NULL,
    bbox_cy             NUMERIC(7,6) NOT NULL,
    bbox_w              NUMERIC(7,6) NOT NULL,
    bbox_h              NUMERIC(7,6) NOT NULL,
    confidence          NUMERIC(5,4) NOT NULL,
    created_at          TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT fk_detections_track_session
        FOREIGN KEY (track_id, video_session_id)
        REFERENCES tracks (id, video_session_id) ON DELETE CASCADE,
    CONSTRAINT chk_detection_frame CHECK (frame_number >= 0 AND timestamp_ms >= 0),
    CONSTRAINT chk_detection_bbox_size CHECK (
        bbox_w > 0 AND bbox_w <= 1 AND bbox_h > 0 AND bbox_h <= 1
    ),
    CONSTRAINT chk_detection_bbox_center CHECK (
        bbox_cx BETWEEN 0 AND 1 AND bbox_cy BETWEEN 0 AND 1
    ),
    CONSTRAINT chk_detection_bbox_containment CHECK (
        bbox_cx - bbox_w / 2 >= 0 AND bbox_cx + bbox_w / 2 <= 1
        AND bbox_cy - bbox_h / 2 >= 0 AND bbox_cy + bbox_h / 2 <= 1
    ),
    CONSTRAINT chk_detection_confidence CHECK (confidence BETWEEN 0 AND 1)
);

CREATE INDEX idx_detections_session_frame ON detections (video_session_id, frame_number);
CREATE INDEX idx_detections_track ON detections (track_id) WHERE track_id IS NOT NULL;

-- =========================================================================
-- 13. SIGHTINGS
-- Only MODEL_CONFIRMED video sessions may be published. Generalized locations
-- only; no exact wildlife coordinates should be collected for public display.
-- =========================================================================
CREATE TABLE sightings (
    id                      UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    video_session_id        UUID NOT NULL UNIQUE REFERENCES video_sessions(id) ON DELETE RESTRICT,
    user_id                 UUID REFERENCES users(id) ON DELETE SET NULL,
    generalized_lat         NUMERIC(8,5),
    generalized_lng         NUMERIC(8,5),
    location_region_label   VARCHAR(150),
    observed_at             TIMESTAMPTZ,
    activity                activity_tag,
    concerns                concern_tag[] NOT NULL DEFAULT '{}',
    consent_given_at        TIMESTAMPTZ,
    consent_version         VARCHAR(30),
    publication_status      publication_status NOT NULL DEFAULT 'DRAFT',
    withdrawn_at            TIMESTAMPTZ,
    withdrawn_by_user_id    UUID REFERENCES users(id) ON DELETE SET NULL,
    withdrawal_reason       TEXT,
    created_at              TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at              TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT chk_sighting_coordinates_pair CHECK (
        (generalized_lat IS NULL AND generalized_lng IS NULL)
        OR (generalized_lat IS NOT NULL AND generalized_lng IS NOT NULL)
    ),
    CONSTRAINT chk_sighting_latitude CHECK (
        generalized_lat IS NULL OR generalized_lat BETWEEN -90 AND 90
    ),
    CONSTRAINT chk_sighting_longitude CHECK (
        generalized_lng IS NULL OR generalized_lng BETWEEN -180 AND 180
    ),
    CONSTRAINT chk_published_sighting_requirements CHECK (
        publication_status <> 'PUBLISHED'
        OR (generalized_lat IS NOT NULL AND generalized_lng IS NOT NULL
            AND observed_at IS NOT NULL AND consent_given_at IS NOT NULL
            AND consent_version IS NOT NULL)
    ),
    CONSTRAINT chk_withdrawn_metadata CHECK (
        publication_status <> 'WITHDRAWN' OR withdrawn_at IS NOT NULL
    )
);

CREATE INDEX idx_sightings_public_map ON sightings (observed_at DESC)
    WHERE publication_status = 'PUBLISHED';
CREATE INDEX idx_sightings_region ON sightings (location_region_label)
    WHERE publication_status = 'PUBLISHED';
CREATE INDEX idx_sightings_user_created ON sightings (user_id, created_at DESC);

        OR (generalized_lat IS NOT NULL AND generalized_lng IS NOT NULL)
    ),
    CONSTRAINT chk_observation_latitude CHECK (
        generalized_lat IS NULL OR generalized_lat BETWEEN -90 AND 90
    ),
    CONSTRAINT chk_observation_longitude CHECK (
        generalized_lng IS NULL OR generalized_lng BETWEEN -180 AND 180
    ),
    CONSTRAINT chk_observation_description_length CHECK (
        description IS NULL OR char_length(description) <= 2000
    ),
    CONSTRAINT chk_observation_submitted_requirements CHECK (
        status <> 'SUBMITTED' OR (submitted_at IS NOT NULL AND observed_at IS NOT NULL)
    )
);

CREATE INDEX idx_marine_observations_user ON marine_observations (user_id, created_at DESC)

-- =========================================================================
-- 20. PREDICTION FEEDBACK
-- Original model output is immutable; feedback proposes a correction for review.
-- =========================================================================
CREATE TABLE prediction_feedback (
    id                        UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    classification_result_id  UUID NOT NULL,
    predicted_class_id        UUID NOT NULL,
    user_id                   UUID REFERENCES users(id) ON DELETE CASCADE,
    anon_token_hash           CHAR(64),
    verdict                   feedback_verdict NOT NULL,
    proposed_class_id         UUID REFERENCES marine_classes(id) ON DELETE RESTRICT,
    comment                   TEXT,
    review_status             feedback_review_status NOT NULL DEFAULT 'PENDING',
    reviewer_user_id          UUID REFERENCES users(id) ON DELETE RESTRICT,
    review_note               TEXT,
    created_at                TIMESTAMPTZ NOT NULL DEFAULT now(),
    reviewed_at               TIMESTAMPTZ,
    CONSTRAINT fk_feedback_result_class
        FOREIGN KEY (classification_result_id, predicted_class_id)
        REFERENCES classification_results (id, predicted_class_id) ON DELETE CASCADE,
    CONSTRAINT chk_feedback_identity CHECK (
        (user_id IS NOT NULL AND anon_token_hash IS NULL)
        OR (user_id IS NULL AND anon_token_hash IS NOT NULL)
    ),
    CONSTRAINT chk_feedback_incorrect_needs_distinct_class CHECK (
        verdict <> 'INCORRECT'
        OR (proposed_class_id IS NOT NULL AND proposed_class_id <> predicted_class_id)
    ),
    CONSTRAINT chk_feedback_proposed_only_when_incorrect CHECK (
        verdict = 'INCORRECT' OR proposed_class_id IS NULL
    ),
    CONSTRAINT chk_feedback_review_consistency CHECK (
        (review_status = 'PENDING') = (reviewed_at IS NULL)
    ),
    CONSTRAINT chk_feedback_review_has_reviewer CHECK (
        review_status = 'PENDING' OR reviewer_user_id IS NOT NULL
    ),
    CONSTRAINT chk_feedback_token_format CHECK (
        anon_token_hash IS NULL OR anon_token_hash ~ '^[0-9a-f]{64}$'
    ),
    CONSTRAINT chk_feedback_comment_length CHECK (
        comment IS NULL OR char_length(comment) <= 1000
    ),
    CONSTRAINT chk_feedback_note_length CHECK (
        review_note IS NULL OR char_length(review_note) <= 1000
    )
);

CREATE UNIQUE INDEX uq_feedback_one_per_anon

-- Model-to-pipeline matching is validated by the backend application.
-- classification_results must use Pipeline A; video_sessions and gating_settings must use Pipeline B.

-- A completed session must use a gating setting belonging to its model.
CREATE OR REPLACE FUNCTION fn_validate_video_gating_setting()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
DECLARE
    v_gating_model_id UUID;
BEGIN
    IF NEW.gating_setting_id IS NOT NULL THEN
        SELECT model_version_id INTO v_gating_model_id
        FROM gating_settings WHERE id = NEW.gating_setting_id;

        IF NOT FOUND THEN
            RAISE EXCEPTION 'Gating setting % does not exist', NEW.gating_setting_id;
        END IF;

        IF v_gating_model_id <> NEW.model_version_id THEN
            RAISE EXCEPTION 'Gating setting model must match video session model';
        END IF;
    END IF;

    IF NEW.processing_status = 'COMPLETED' AND NEW.gating_setting_id IS NULL THEN
        RAISE EXCEPTION 'Completed video sessions require a gating setting';
    END IF;

    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_video_gating_setting_consistency
    BEFORE INSERT OR UPDATE
    ON video_sessions
    FOR EACH ROW EXECUTE FUNCTION fn_validate_video_gating_setting();

-- Do not mutate the identity/version fields of a model version after it is used.
CREATE OR REPLACE FUNCTION fn_lock_used_model_version()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
    IF (NEW.pipeline IS DISTINCT FROM OLD.pipeline
        OR NEW.model_name IS DISTINCT FROM OLD.model_name
        OR NEW.version_tag IS DISTINCT FROM OLD.version_tag)
       AND (
          EXISTS (SELECT 1 FROM classification_results WHERE model_version_id = OLD.id)
          OR EXISTS (SELECT 1 FROM video_sessions WHERE model_version_id = OLD.id)
          OR EXISTS (SELECT 1 FROM gating_settings WHERE model_version_id = OLD.id)
       )
    THEN
        RAISE EXCEPTION 'Cannot change pipeline/name/version tag of a model version that is already referenced';
    END IF;
    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_lock_used_model_version
    BEFORE UPDATE ON model_versions
    FOR EACH ROW EXECUTE FUNCTION fn_lock_used_model_version();

-- Lock gating configuration fields after a session references the row.
-- is_active can change so a new setting can become active.
CREATE OR REPLACE FUNCTION fn_lock_used_gating_settings()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
    IF EXISTS (SELECT 1 FROM video_sessions WHERE gating_setting_id = OLD.id)
       AND (
          NEW.model_version_id IS DISTINCT FROM OLD.model_version_id
          OR NEW.confidence_threshold IS DISTINCT FROM OLD.confidence_threshold
          OR NEW.temporal_window_ms IS DISTINCT FROM OLD.temporal_window_ms
          OR NEW.minimum_detection_count IS DISTINCT FROM OLD.minimum_detection_count
          OR NEW.aggregation_method IS DISTINCT FROM OLD.aggregation_method
          OR NEW.frame_sampling_fps IS DISTINCT FROM OLD.frame_sampling_fps
          OR NEW.notes IS DISTINCT FROM OLD.notes
       )
    THEN
        RAISE EXCEPTION 'Cannot change configuration of gating setting % after it has been used; insert a new row', OLD.id;
    END IF;
    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_lock_used_gating_settings
    BEFORE UPDATE ON gating_settings
    FOR EACH ROW EXECUTE FUNCTION fn_lock_used_gating_settings();

-- Published sightings require complete metadata and a confirmed session.
-- Locking the session row prevents a concurrent gate-decision change race.
CREATE OR REPLACE FUNCTION fn_enforce_sighting_publish_rule()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
DECLARE
    v_gate_decision decision_status;
BEGIN
    IF NEW.publication_status = 'PUBLISHED' THEN
        SELECT gate_decision INTO v_gate_decision
        FROM video_sessions
        WHERE id = NEW.video_session_id
        FOR UPDATE;

        IF NOT FOUND OR v_gate_decision IS DISTINCT FROM 'MODEL_CONFIRMED' THEN
            RAISE EXCEPTION
                'Sighting % cannot be published unless its video session is MODEL_CONFIRMED',
                NEW.id;
        END IF;
    END IF;
    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_sightings_publish_rule
    BEFORE INSERT OR UPDATE ON sightings
    FOR EACH ROW EXECUTE FUNCTION fn_enforce_sighting_publish_rule();

-- Once a published sighting relies on a decision/configuration, do not mutate
-- the decision or model/gating references. Withdraw the sighting to retract it.
CREATE OR REPLACE FUNCTION fn_lock_published_session_decision()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
    IF EXISTS (
        SELECT 1 FROM sightings
        WHERE video_session_id = OLD.id AND publication_status = 'PUBLISHED'
    ) THEN
        IF NEW.gate_decision IS DISTINCT FROM OLD.gate_decision
           OR NEW.model_version_id IS DISTINCT FROM OLD.model_version_id
           OR NEW.gating_setting_id IS DISTINCT FROM OLD.gating_setting_id
        THEN
            RAISE EXCEPTION
                'Cannot change gate decision or model/gating configuration while a published sighting depends on session %; withdraw the sighting first',
                OLD.id;
        END IF;
    END IF;
    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_video_sessions_lock_published_decision
    BEFORE UPDATE ON video_sessions
    FOR EACH ROW EXECUTE FUNCTION fn_lock_published_session_decision();

-- Prediction feedback must be tied to the owner of the classification result.
CREATE OR REPLACE FUNCTION fn_validate_prediction_feedback_owner()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
DECLARE
    v_result_user_id UUID;
    v_result_anon_token_hash CHAR(64);
BEGIN
    SELECT user_id, anon_token_hash
      INTO v_result_user_id, v_result_anon_token_hash
    FROM classification_results
    WHERE id = NEW.classification_result_id
      AND predicted_class_id = NEW.predicted_class_id;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Classification result and predicted class do not match';
    END IF;

    IF NEW.user_id IS NOT NULL THEN
        IF v_result_user_id IS DISTINCT FROM NEW.user_id
           OR v_result_anon_token_hash IS NOT NULL THEN
            RAISE EXCEPTION 'Feedback user must own the classification result';
        END IF;
    ELSE
        IF v_result_user_id IS NOT NULL
           OR v_result_anon_token_hash IS DISTINCT FROM NEW.anon_token_hash THEN
            RAISE EXCEPTION 'Anonymous feedback token must match the classification result owner token';
        END IF;
    END IF;

    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_prediction_feedback_owner
    BEFORE INSERT OR UPDATE
    ON prediction_feedback
    FOR EACH ROW EXECUTE FUNCTION fn_validate_prediction_feedback_owner();

-- Only admins may review feedback. A non-pending review requires a reviewer.
CREATE OR REPLACE FUNCTION fn_validate_feedback_reviewer()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
DECLARE
    v_role user_role;
BEGIN
    IF NEW.review_status <> 'PENDING' THEN
        IF NEW.reviewer_user_id IS NULL THEN
            RAISE EXCEPTION 'A reviewed feedback item requires reviewer_user_id';
        END IF;

        SELECT role INTO v_role FROM users WHERE id = NEW.reviewer_user_id;
        IF NOT FOUND OR v_role <> 'ADMIN' THEN
            RAISE EXCEPTION 'Only an ADMIN may review prediction feedback';
        END IF;
    END IF;
    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_prediction_feedback_reviewer
    BEFORE INSERT OR UPDATE
    ON prediction_feedback
    FOR EACH ROW EXECUTE FUNCTION fn_validate_feedback_reviewer();

-- =========================================================================
-- RAG SOURCE INVALIDATION 
-- On source edits/deletes, deactivate stale chunks. The application/admin must
-- regenerate embeddings and reactivate fresh chunks after curated content changes.
-- =========================================================================

CREATE OR REPLACE FUNCTION fn_species_embedding_change()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
    IF TG_OP = 'DELETE' THEN
        UPDATE knowledge_chunks
        SET is_active = false, updated_at = now()
        WHERE source_type = 'SPECIES' AND source_id = OLD.id;
        RETURN OLD;
    ELSIF TG_OP = 'UPDATE' THEN
        IF NEW.is_active IS DISTINCT FROM OLD.is_active
           OR NEW.common_name_en IS DISTINCT FROM OLD.common_name_en
           OR NEW.common_name_si IS DISTINCT FROM OLD.common_name_si
           OR NEW.common_name_ta IS DISTINCT FROM OLD.common_name_ta
           OR NEW.scientific_name IS DISTINCT FROM OLD.scientific_name
           OR NEW.marine_class_id IS DISTINCT FROM OLD.marine_class_id
           OR NEW.description_en IS DISTINCT FROM OLD.description_en
           OR NEW.description_si IS DISTINCT FROM OLD.description_si
           OR NEW.description_ta IS DISTINCT FROM OLD.description_ta
           OR NEW.habitat_en IS DISTINCT FROM OLD.habitat_en
           OR NEW.habitat_si IS DISTINCT FROM OLD.habitat_si
           OR NEW.habitat_ta IS DISTINCT FROM OLD.habitat_ta
           OR NEW.distribution_en IS DISTINCT FROM OLD.distribution_en
           OR NEW.distribution_si IS DISTINCT FROM OLD.distribution_si
           OR NEW.distribution_ta IS DISTINCT FROM OLD.distribution_ta
           OR NEW.threats_en IS DISTINCT FROM OLD.threats_en
           OR NEW.threats_si IS DISTINCT FROM OLD.threats_si
           OR NEW.threats_ta IS DISTINCT FROM OLD.threats_ta
           OR NEW.facts IS DISTINCT FROM OLD.facts
           OR NEW.conservation_status IS DISTINCT FROM OLD.conservation_status
           OR NEW.is_sri_lankan IS DISTINCT FROM OLD.is_sri_lankan
        THEN
            UPDATE knowledge_chunks
            SET is_active = false, updated_at = now()
            WHERE source_type = 'SPECIES' AND source_id = NEW.id;
        END IF;
        RETURN NEW;
    END IF;
    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_marine_species_embedding_change
    AFTER UPDATE OR DELETE ON marine_species
    FOR EACH ROW EXECUTE FUNCTION fn_species_embedding_change();
